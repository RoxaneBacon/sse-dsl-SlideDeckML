Highlight Mode: all

Highlights all lines at once:

function greet(name) {
    console.log("Hello, " + name);
    return true;
}

Highlight Mode: none

No highlighting:

def calculate(x, y):
    result = x + y
    return result

Highlight Mode: line-by-line

Reveals each line progressively:

interface User {
    id: number;
    name: string;
}

const user: User = {
    id: 1,
    name: "Alice"
};

Highlight Mode: block

Highlights code blocks separated by empty lines:

public class Calculator {
    private int value;

    public void add(int x) {
        value += x;
    }

    public int getValue() {
        return value;
    }
}

Highlight Mode: function

Detects and highlights functions:

def factorial(n):
    if n <= 1:
        return 1
    return n * factorial(n - 1)

def fibonacci(n):
    if n <= 1:
        return n
    return fibonacci(n - 1) + fibonacci(n - 2)

Highlight Mode: class

Detects and highlights classes:

class Person {
    constructor(name, age) {
        this.name = name;
        this.age = age;
    }
    
    introduce() {
        return `I'm ${this.name}, ${this.age} years old`;
    }
}

class Student extends Person {
    constructor(name, age, grade) {
        super(name, age);
        this.grade = grade;
    }
}
