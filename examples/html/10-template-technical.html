Service Communication

API Gateway Pattern

class APIGateway {
    constructor(services) {
        this.services = services;
        this.loadBalancer = new LoadBalancer();
    }
    
    async route(request) {
        const service = this.services[request.path];
        const instance = this.loadBalancer.getNext(service);
        return await instance.handle(request);
    }
}

Service Implementation

Microservice Structure

interface Service {
    name: string;
    port: number;
    health: () => Promise<boolean>;
}

class OrderService implements Service {
    name = 'order-service';
    port = 3001;
    
    async health() {
        return this.database.isConnected();
    }
    
    async createOrder(data: OrderData) {
        const order = await this.repository.create(data);
        await this.eventBus.publish('order.created', order);
        return order;
    }
}

Event-Driven Architecture

Asynchronous Communication

// Publisher
eventBus.publish('order.created', {
    orderId: '12345',
    userId: 'user-001',
    amount: 99.99
});

// Subscriber
eventBus.subscribe('order.created', 
    async (event) => {
        await sendConfirmationEmail(event);
        await updateInventory(event);
    }
);

Best Practices

Key Recommendations

  1. Single Responsibility - Each service does one thing well
  2. API Versioning - Maintain backward compatibility
  3. Circuit Breakers - Prevent cascade failures
  4. Distributed Tracing - Track requests across services
  5. Database per Service - Avoid shared databases
  6. Automated Testing - Unit, integration, and E2E tests
